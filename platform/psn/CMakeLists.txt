cmake_minimum_required(VERSION 3.24)
project(psn_host LANGUAGES C)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The PS4 guest adapter requires a 64-bit host")
endif()

find_package(Threads REQUIRED)
add_library(psn-host STATIC psn_host.c psn_identity.c psn_webapi.c)
set_target_properties(psn-host PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_include_directories(psn-host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(psn-host PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(psn-host PRIVATE -Wall -Wextra -Werror)
target_link_libraries(psn-host PUBLIC Threads::Threads ${CMAKE_DL_LIBS})

option(PSN_HOST_TESTS "Build host contract tests (no game, GPU or network)" ${PROJECT_IS_TOP_LEVEL})
option(PSN_BRIDGE_BACKEND "Build the optional authenticated loopback bridge backend" OFF)
if(PSN_BRIDGE_BACKEND)
    if(PSN_HOST_TESTS)
        enable_testing()
    endif()
    enable_language(CXX)
    add_subdirectory(backends)
endif()
if(PSN_HOST_TESTS)
    enable_testing()
    foreach(part identity webapi)
        add_executable(psn-${part}-test tests/test_psn_${part}.c psn_${part}.c)
        target_include_directories(psn-${part}-test PRIVATE .)
        target_compile_definitions(psn-${part}-test PRIVATE _POSIX_C_SOURCE=200809L)
        target_compile_options(psn-${part}-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
        target_link_libraries(psn-${part}-test PRIVATE Threads::Threads)
        set_target_properties(psn-${part}-test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
        add_test(NAME psn-${part} COMMAND psn-${part}-test)
        add_test(NAME psn-${part}-arbitrary-user COMMAND psn-${part}-test 70001)
        add_test(NAME psn-${part}-zero-user COMMAND psn-${part}-test 0)
        set_tests_properties(psn-${part} psn-${part}-arbitrary-user psn-${part}-zero-user
            PROPERTIES TIMEOUT 20)
    endforeach()
    add_library(psn-backend-fixture MODULE tests/psn_backend_fixture.c)
    set_target_properties(psn-backend-fixture PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    target_compile_options(psn-backend-fixture PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    add_executable(psn-backend-test tests/test_psn_backend.c psn_host.c)
    target_include_directories(psn-backend-test PRIVATE .)
    target_compile_definitions(psn-backend-test PRIVATE _POSIX_C_SOURCE=200809L)
    target_compile_options(psn-backend-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    target_link_libraries(psn-backend-test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    set_target_properties(psn-backend-test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
    foreach(case default empty unconfigured-users relative missing bad-abi bad-size bad-capabilities missing-function
            query-null open-error open-null open-partial good good-arbitrary-user overflow invalid-event invalid-invite
            invalid-user invalid-state invalid-push oversized-push event-error state-error push-error)
        add_test(NAME psn-backend-${case} COMMAND psn-backend-test $<TARGET_FILE:psn-backend-fixture> ${case})
        set_tests_properties(psn-backend-${case} PROPERTIES TIMEOUT 10)
    endforeach()
endif()
